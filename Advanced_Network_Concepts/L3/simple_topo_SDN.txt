from mininet.net import Mininet
from mininet.node import OVSController
from mininet.cli import CLI
from mininet.log import setLogLevel
from mininet.node import RemoteController

def simple_topology():
    net = Mininet(controller = None)
    #Skapar hosts i nätet
    h1 = net.addHost('h1')
    h2 = net.addHost('h2')
    h3 = net.addHost('h3')

    #Lägger till switchar i nätet
    s1 = net.addSwitch('s1')
    s2 = net.addSwitch('s2')
    s3 = net.addSwitch('s3')

    net.addController('c0', controller = RemoteController, ip = '127.0.0.1', port=6633)
    #Lägger till hosts i switcharna
    net.addLink(h1, s1)
    net.addLink(h2, s2)
    net.addLink(h3, s3)
    #Länkar switcharna
    print("Linking switches")
    net.addLink(s1, s2)
    net.addLink(s2, s3)
    net.addLink(s1, s3)
    net.start()
    CLI(net)
    net.stop()

if __name__ == '__main__':
    #setLogLevel('info')
    simple_topology()
